{
  "name": "My AI agent tests",
  "description": "Smoke tests for my agent",
  "test_cases": [
    {
      "name": "Mentions the refund window",
      "input": {
        "message": "What is the refund deadline?",
        "mock_response": { "content": "Eligible items can be returned within 30 days." }
      },
      "required_keywords": ["30 days"],
      "forbidden_keywords": ["guaranteed"],
      "grading_rules": { "exactMatch": false }
    }
  ]
}
